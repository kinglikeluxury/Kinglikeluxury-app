-- Single manual schema for the isolated employee webhook + archive review.
-- Never run at application startup or against Neon/production.
CREATE TABLE whatsapp_employee_numbers (
  phone_number_id TEXT PRIMARY KEY,
  waba_id TEXT NOT NULL,
  employee_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  is_active BOOLEAN NOT NULL DEFAULT FALSE,
  business_authorized_at TIMESTAMPTZ,
  authorization_reference TEXT,
  authorized_by INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  activated_by INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  activated_at TIMESTAMPTZ,
  coexistence_enabled BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (phone_number_id,waba_id,employee_id),
  UNIQUE (phone_number_id,employee_id)
);

CREATE TABLE whatsapp_employee_consents (
  phone_number_id TEXT NOT NULL,
  employee_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  status TEXT NOT NULL CHECK (status IN ('consented','revoked')),
  consented_at TIMESTAMPTZ,
  revoked_at TIMESTAMPTZ,
  last_actor_id INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  PRIMARY KEY (phone_number_id,employee_id),
  FOREIGN KEY (phone_number_id,employee_id)
    REFERENCES whatsapp_employee_numbers(phone_number_id,employee_id) ON DELETE RESTRICT
);

CREATE TABLE whatsapp_employee_supervisor_grants (
  supervisor_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  employee_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  can_read BOOLEAN NOT NULL DEFAULT FALSE,
  granted_by INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  revoked_at TIMESTAMPTZ,
  revoked_by INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  PRIMARY KEY (supervisor_id,employee_id),
  CHECK (supervisor_id <> employee_id)
);

CREATE TABLE whatsapp_employee_conversations (
  id BIGSERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  phone_number_id TEXT NOT NULL,
  waba_id TEXT NOT NULL,
  contact_wa_id TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (employee_id,phone_number_id,contact_wa_id),
  FOREIGN KEY (phone_number_id,employee_id)
    REFERENCES whatsapp_employee_numbers(phone_number_id,employee_id) ON DELETE RESTRICT
);
CREATE INDEX whatsapp_employee_conversations_employee_updated_idx
  ON whatsapp_employee_conversations(employee_id,updated_at DESC);

CREATE TABLE whatsapp_employee_events (
  id BIGSERIAL PRIMARY KEY,
  conversation_id BIGINT NOT NULL REFERENCES whatsapp_employee_conversations(id) ON DELETE RESTRICT,
  employee_id INTEGER NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  meta_event_key TEXT NOT NULL UNIQUE,
  meta_message_id TEXT NOT NULL,
  event_kind TEXT NOT NULL CHECK (event_kind IN ('message','status')),
  direction TEXT NOT NULL CHECK (direction IN ('inbound','outbound','status')),
  event_status TEXT,
  message_type TEXT,
  message_text TEXT,
  source_field TEXT NOT NULL CHECK (source_field IN ('messages','smb_message_echoes')),
  occurred_at TIMESTAMPTZ,
  received_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  expires_at TIMESTAMPTZ NULL,
  media_mime_type TEXT,
  media_size_bytes BIGINT CHECK (media_size_bytes IS NULL OR media_size_bytes >= 0),
  media_sha256 TEXT CHECK (media_sha256 IS NULL OR media_sha256 ~ '^[a-f0-9]{64}$')
);
CREATE INDEX whatsapp_employee_events_conversation_received_idx
  ON whatsapp_employee_events(conversation_id,received_at DESC);
CREATE INDEX whatsapp_employee_events_expiry_idx ON whatsapp_employee_events(expires_at);

CREATE TABLE whatsapp_employee_audit (
  id BIGSERIAL PRIMARY KEY,
  actor_id INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  employee_id INTEGER REFERENCES users(id) ON DELETE RESTRICT,
  action TEXT NOT NULL,
  phone_number_id TEXT,
  waba_id TEXT,
  meta_event_key TEXT,
  outcome TEXT NOT NULL DEFAULT 'recorded',
  details JSONB NOT NULL DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (action,meta_event_key,outcome)
);
CREATE INDEX whatsapp_employee_audit_scope_created_idx
  ON whatsapp_employee_audit(employee_id,created_at DESC);
